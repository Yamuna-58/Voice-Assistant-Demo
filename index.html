function processCommand(command) { command = command.toLowerCase().trim(); // Remove punctuation command = command.replace(/[.,!?]/g, ""); console.log("Recognized command:", command); // Greeting if (/\b(hello|hi|hey)\b/.test(command)) { speak("Hello! How can I help you?"); return; } // Reminder if (command.includes("remind me") || command.includes("set reminder")) { setReminder(command); return; } // Timer if (command.includes("timer") || command.includes("countdown")) { setTimer(command); return; } // Wikipedia if (command.includes("wikipedia")) { wikipediaSearch(command); return; } // YouTube search if (command.includes("search youtube") || command.includes("youtube search") || command.includes("search on youtube")) { youtubeSearch(command); return; } // Weather if (command.includes("weather")) { weatherSearch(command); return; } // Jokes if (command.includes("joke")) { tellJoke(); return; } // Gmail if (command.includes("gmail")) { speak("Opening Gmail."); window.open("https://mail.google.com", "_blank"); return; } // ChatGPT if (command.includes("chatgpt") || command.includes("chat gpt")) { speak("Opening ChatGPT."); window.open("https://chatgpt.com", "_blank"); return; } // GitHub if (command.includes("github")) { speak("Opening GitHub."); window.open("https://github.com", "_blank"); return; } // Google Maps if (command.includes("google maps") || command.includes("open maps")) { speak("Opening Google Maps."); window.open("https://maps.google.com", "_blank"); return; } // YouTube if (command.includes("youtube")) { speak("Opening YouTube."); window.open("https://www.youtube.com", "_blank"); return; } // Google if (command.includes("google")) { speak("Opening Google."); window.open("https://www.google.com", "_blank"); return; } // System information if (command.includes("system information") || command.includes("system details")) { systemInformation(); return; } // Search if (command.startsWith("search") || command.startsWith("google")) { googleSearch(command); return; } // Time if (command.includes("time")) { speak("The current time is " + new Date().toLocaleTimeString()); return; } // Date if (command.includes("date") || command.includes("today")) { speak("Today's date is " + new Date().toLocaleDateString()); return; } // Help if (command.includes("help")) { speak("You can ask me for time, date, timer, reminders, Google search, YouTube, Wikipedia, weather, Gmail, ChatGPT, GitHub, Maps and jokes."); return; } // Exit if (command.includes("exit") || command.includes("quit") || command.includes("goodbye")) { stopAssistant(); speak("Goodbye! Have a nice day."); return; } speak("Sorry, I don't know that command. Please try again."); }
